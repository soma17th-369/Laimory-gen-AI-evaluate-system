아래 파일은 현재 전체 스냅샷이다. 기존 코드도 포함되므로 파일 전체가 이번에 신규 작성된 것으로 해석하면 안 된다. 이번 작업의 추가·수정 범위는 상세 보고서의 파일별 표를 따른다.

## app/analysis/calibration.py

[app/analysis/calibration.py](../../app/analysis/calibration.py)

````python
"""세부 기준 판정 → 고정 환산. LLM은 점수 대신 통과 건수와 근거를 반환한다."""

from __future__ import annotations

import re
from fractions import Fraction
from pathlib import Path

from pydantic import BaseModel, ConfigDict, Field, create_model, model_validator

from app.analysis import stages

POLICY_VERSION = "criteria-ratio-v1"
ASSETS = Path(__file__).resolve().parents[2] / "assets" / "rubrics" / "calibration"


def checks(text: str) -> tuple[str, ...]:
    """최상위 bullet 하나(하위 조건 포함)를 한 기준으로 고정한다.

    User Memory 고정 필드 표는 데이터 행별로 나눈다. 설명·참조 표는 원문 프롬프트에 남으며
    별도 배점 기준으로 중복 계산하지 않는다. 자유문 편집본은 전체를 한 기준으로 사용한다.
    """
    lines = text.splitlines()
    starts = [i for i, line in enumerate(lines) if re.match(r"^[ ]{0,1}[-*] ", line)]
    if starts:
        result = []
        for index, start in enumerate(starts):
            end = starts[index + 1] if index + 1 < len(starts) else len(lines)
            body = []
            for line in lines[start:end]:
                if line.strip().startswith(("코드 측정", "#", "|")):
                    break
                body.append(line)
            result.append("\n".join(body).strip().removeprefix("- ").removeprefix("* "))
        return tuple(result)
    rows = [line.strip() for line in lines if line.strip().startswith("| `")]
    return tuple(rows) if rows else (text.strip(),)


def definitions(spec: stages.StageSpec) -> dict[str, tuple[str, ...]]:
    return {entry.item.key: checks(entry.text) for entry in stages.criteria(spec)}


class CheckAssessment(BaseModel):
    model_config = ConfigDict(extra="forbid")

    total: int = Field(ge=0, strict=True, description="이 기준에 해당하는 평가 단위 수. 없을 때만 0")
    passed: int = Field(ge=0, strict=True, description="모든 하위 조건을 충족한 평가 단위 수")
    failedRefs: list[str] = Field(description="실패 단위별 고유 ID 또는 field 경로. total-passed개")
    reason: str = Field(min_length=1, description="통과·실패 판정 근거. total=0이면 입력 부재 사유")

    @model_validator(mode="after")
    def consistent_counts(self):
        if self.passed > self.total:
            raise ValueError("통과 수는 평가 수보다 클 수 없습니다")
        if len(self.failedRefs) != self.total - self.passed:
            raise ValueError("실패 단위마다 근거 위치가 하나씩 필요합니다")
        if len(set(self.failedRefs)) != len(self.failedRefs) or any(not ref.strip() for ref in self.failedRefs):
            raise ValueError("실패 근거 위치는 비어 있거나 중복될 수 없습니다")
        if not self.reason.strip():
            raise ValueError("판정 근거는 비어 있을 수 없습니다")
        return self


def result_model(spec: stages.StageSpec) -> type[BaseModel]:
    """편집된 확인 기준까지 스키마에 고정한다. 기준 누락·임의 기준 추가를 거부한다."""
    from app.analysis.stage_schema import Alert, Deduction

    item_fields = {}
    for key, texts in definitions(spec).items():
        check_model = create_model(
            f"{spec.key}_{key}_Checks",
            __config__=ConfigDict(extra="forbid"),
            **{f"c{i:02}": (CheckAssessment, ...) for i in range(1, len(texts) + 1)},
        )
        item_fields[key] = (check_model, ...)
    items = create_model(f"{spec.key}_Assessments", __config__=ConfigDict(extra="forbid"), **item_fields)
    return create_model(
        f"{spec.key}_CalibratedJudge",
        __config__=ConfigDict(extra="forbid"),
        items=(items, ...),
        deductions=(list[Deduction], ...),
        alerts=(list[Alert], ...),
        summary=(str, ...),
    )


def points_from_checks(points: int, assessments: list[CheckAssessment]) -> tuple[int, bool]:
    ratios = [Fraction(row.passed, row.total) for row in assessments if row.total]
    if not ratios:
        return 0, False
    value = points * sum(ratios) / len(ratios)
    # 항목별 정수 점수 계약 유지. 정확한 유리수 계산 후 .5 이상 올림(은행가 반올림 아님).
    rounded = (2 * value.numerator + value.denominator) // (2 * value.denominator)
    return rounded, True


def score(spec: stages.StageSpec, card: BaseModel, *, usage: dict | None = None):
    from app.analysis import stage_schema

    payload = card.model_dump()
    item_scores = {}
    for item in spec.items:
        item_card = getattr(card.items, item.key)
        rows = [getattr(item_card, key) for key in type(item_card).model_fields]
        points, applied = points_from_checks(item.points, rows)
        reason = "\n".join(
            f"c{i:02}: {row.passed}/{row.total} 통과 — {row.reason}"
            for i, row in enumerate(rows, 1)
        )
        item_scores[item.key] = {"score": points, "applied": applied, "reason": reason}
    legacy_shape = stage_schema.scorecard_model(spec.key).model_validate({**payload, "items": item_scores})
    return stage_schema.tally(
        spec, legacy_shape,
        extras={"usage": usage or {}, "scoringPolicy": POLICY_VERSION, "checks": payload["items"]},
    )


def prompt(spec: stages.StageSpec) -> str:
    blocks = [(ASSETS / "rules.md").read_text(encoding="utf-8")]
    by_key = definitions(spec)
    for item in spec.items:
        texts = by_key[item.key]
        blocks.append(f"### 판정 ID: {item.key} — {item.points}점")
        blocks.extend(f"- c{i:02}: {text}" for i, text in enumerate(texts, 1))
        examples = [
            points_from_checks(item.points, [CheckAssessment(total=5, passed=n, failedRefs=[f"E{i}" for i in range(5-n)], reason="합성 예시")])[0]
            for n in (5, 4, 0)
        ]
        blocks.append(
            f"환산 예시(모든 적용 기준의 통과율이 같은 합성 사례): 5/5 → {examples[0]}점, "
            f"4/5 → {examples[1]}점, 0/5 → {examples[2]}점. 모든 기준 0/0 → N/A."
        )
    # 편집본에는 원래 의미와 달라진 항목이 있을 수 있어 원본 전용 예시를 섞지 않는다.
    if not stages.is_edited(spec.key):
        blocks.append((ASSETS / f"{spec.key}.md").read_text(encoding="utf-8"))
    return "\n\n".join(blocks)

````

````python
"""형식이 확인된 질문 입력·출력에만 적용하는 결정론 검사."""

from __future__ import annotations

from collections import Counter

from app.analysis import stages
from app.analysis.evidence import StageEvidence, parse_json_array


def confirmed_checks(evidence: StageEvidence) -> dict:
    """읽을 수 없는 자료는 측정값을 만들지 않는다. 편집본은 다른 규칙일 수 있어 제외한다."""
    if evidence.key != "question" or stages.is_edited(evidence.key):
        return {}
    sections = {section.title: section.payload for section in evidence.sections}
    inputs = parse_json_array(sections.get("입력 event"))
    outputs = parse_json_array(sections.get("출력 질문"))
    if inputs is None or outputs is None:
        return {}
    ids = [row.get("clientEventId") for row in inputs]
    if any(not isinstance(key, str) or not key.strip() for key in ids) or len(set(ids)) != len(ids):
        return {}
    if any(not isinstance(row.get("clientEventId"), str) for row in outputs):
        return {}
    counts = Counter(row["clientEventId"] for row in outputs if isinstance(row.get("question"), str) and row["question"].strip())
    failed = [key for key in ids if counts[key] != 1]
    coverage = {
        "total": len(ids), "passed": len(ids) - len(failed), "failedRefs": failed,
        "reason": f"코드 확인: 입력 ID {len(ids)}개 중 질문이 정확히 1개인 ID {len(ids)-len(failed)}개. 유형별로도 동일 입력 ID를 대조함.",
    }
    lengths = [len(row["question"].strip()) if isinstance(row.get("question"), str) else 0 for row in outputs]
    failed_lengths = [f"questions[{i}].question" for i, length in enumerate(lengths) if not 1 <= length <= 40]
    present = {row["clientEventId"] for row in outputs}
    missing = [key for key in ids if key not in present]
    failed_lengths.extend(f"missing:{key}.question" for key in missing)
    length_total = len(outputs) + len(missing)
    return {
        "coverage": {"c01": coverage, "c02": dict(coverage)},
        "expression": {"c02": {
            "total": length_total, "passed": length_total - len(failed_lengths), "failedRefs": failed_lengths,
            "reason": "코드 확인: 앞뒤 공백 제거 후 Unicode 문자 수 1~40자. 빈 질문·누락은 실패. "
                      f"출력별 글자 수={lengths}, 누락={len(missing)}개.",
        }},
    }
````

## app/analysis/stage_judge.py

[app/analysis/stage_judge.py](../../app/analysis/stage_judge.py)

````python
"""단계별 LLM-as-a-Judge 실행.

단계 하나마다 (1) 그 단계의 근거만 자르고 (2) 메트릭 문서 원문을 채점 기준으로 주고 (3) 항목별
세부 기준별 통과 건수·이유를 구조화 출력으로 받는다. 항목 점수는 calibration.score,
총점은 [stage_schema.tally][app.analysis.stage_schema.tally]가 계산한다.

채점 기준의 정본은 **현재 채점 항목**이다(메트릭 문서 기본값 + 설정 화면에서 고친 편집본,
→ [app.analysis.stages.get][]). 프롬프트는 채점할 때마다 그 항목으로 다시 만들어지므로, 항목을
고치면 다음 채점부터 바로 반영된다. 레지스트리(`judge-stage-<stageKey>`)에는 **실제로 쓴 본문을
기록**해 두어, 나중에 어떤 기준으로 매긴 점수인지 되짚을 수 있게 한다.
"""

from __future__ import annotations

from concurrent.futures import ThreadPoolExecutor
from typing import Any, Callable

from app.analysis import calibration, evidence, stage_schema, stages
from app.analysis.evidence import StageEvidence
from app.analysis.providers import get_llm_provider
from app.analysis.stage_schema import StageScore
from app.analysis.stages import StageSpec
from app.prompts import registry

RUBRIC_PREFIX = "judge-stage-"

_JUDGE_ROLE = """\
당신은 Laimory 생성형 AI 파이프라인의 **{label}** 결과를 채점하는 엄정한 채점자다.

- 아래 채점 기준 문서에 있는 **모든 항목**의 세부 기준을 판정하고, 기준마다 이유를 남긴다. 만점이어도
  무엇을 확인해 문제가 없었는지 적는다.
- 오직 주어진 근거(이 단계의 입력·출력)만으로 판단한다. 근거에 없는 사실을 지어내지 않는다.
- **감점(0점)과 미평가(N/A)는 다르다.** 그 항목을 평가할 입력이 근거에 하나도 없으면 0점을 주지 말고
  해당 기준을 `total=0, passed=0` 으로 둔다. 입력은 있는데 필수 결과가 없거나 틀리면 실패다.
  세부 판정에 따라 항목 점수와 applied 및 총점은 코드가 계산한다.
- 감점은 `deductions` 에 입력 rawId·출력 위치(field 경로)와 함께 적는다.
- 입력에서 확인되지 않는 사실, 필수 반영 source 누락, 민감정보 검출은 점수와 별개로 `alerts` 에 적는다.
- 모든 텍스트는 한국어로 쓰고, 지정된 JSON 스키마로만 답한다.

아래는 채점 기준 문서 원문이다. 요약본이 아니라 정본이므로 여기 적힌 확인 기준을 그대로 적용한다.

"""


def rubric_name(stage_key: str) -> str:
    """단계별 채점 기준의 레지스트리 이름."""
    return f"{RUBRIC_PREFIX}{stage_key}"


def default_rubric(spec: StageSpec) -> str:
    """채점자 지시 + 공통 규칙 + 채점 항목(키·배점·확인 기준) + 점수 작성 형식."""
    return _JUDGE_ROLE.format(label=spec.label) + stages.rubric_text(spec) + "\n\n" + calibration.prompt(spec)


def system_prompt(stage_key: str) -> str:
    """지금 채점에 쓸 기준 본문. 현재 채점 항목으로 매번 새로 만든다."""
    return default_rubric(stages.get(stage_key))


def register_rubric(stage_key: str, content: str, *, note: str = "", source: str = "judge") -> dict:
    """채점에 쓴 기준을 버전으로 확정한다(같은 본문이면 기존 버전 재사용)."""
    return registry.get_or_create(rubric_name(stage_key), content, note=note, source=source)


def score_evidence(
    stage_evidence: StageEvidence,
    *,
    system_prompt_text: str | None = None,
) -> StageScore:
    """근거 하나를 채점한다. 근거가 없으면 LLM 을 부르지 않고 N/A 로 돌려준다."""
    spec = stage_evidence.stage
    if not stage_evidence.applicable:
        return stage_schema.skipped(
            spec, stage_evidence.skip_reason or "채점할 근거가 없습니다.", usage=stage_evidence.usage
        )
    from app.analysis.deterministic_checks import confirmed_checks

    confirmed = confirmed_checks(stage_evidence)
    user_prompt = evidence.build_user_prompt(stage_evidence)
    if confirmed:
        user_prompt += "\n\n[확정 판정 — 코드 계산]\n" + evidence._as_text(confirmed)
    user_prompt += "\n\n점수·applied 대신 지정된 모든 기준 ID의 total·passed·failedRefs·reason을 반환하라."
    card = get_llm_provider().evaluate(
        system_prompt=system_prompt_text or system_prompt(spec.key),
        user_prompt=user_prompt,
        result_model=calibration.result_model(spec),
    )
    for key, values in confirmed.items():
        for check_id, value in values.items():
            setattr(getattr(card.items, key), check_id, calibration.CheckAssessment.model_validate(value))
    return calibration.score(spec, card, usage=stage_evidence.usage)


def score_stage(
    trace_json: dict,
    stage_key: str,
    *,
    system_prompt_text: str | None = None,
) -> StageScore:
    """trace.json 에서 단계 하나를 채점한다."""
    return score_evidence(
        evidence.build(trace_json, stage_key), system_prompt_text=system_prompt_text
    )


def score_stages(
    trace_json: dict,
    stage_keys: list[str],
    *,
    workers: int = 3,
    prompts: dict[str, str] | None = None,
    on_done: Callable[[str, StageScore | None, Exception | None], None] | None = None,
) -> dict[str, StageScore]:
    """여러 단계를 병렬로 채점한다. 한 단계가 실패해도 나머지는 계속한다.

    `on_done` 은 단계가 끝날 때마다 (stageKey, 결과, 예외) 로 불린다. 화면 진행 표시에 쓴다.
    """
    results: dict[str, StageScore] = {}
    evidences = {key: evidence.build(trace_json, key) for key in stage_keys}
    # 근거가 없는 단계는 LLM 호출 없이 먼저 정리한다.
    pending = []
    for key, item in evidences.items():
        if not item.applicable:
            results[key] = stage_schema.skipped(
                item.stage, item.skip_reason or "채점할 근거가 없습니다.", usage=item.usage
            )
            if on_done is not None:
                on_done(key, results[key], None)
        else:
            pending.append(key)

    if not pending:
        return results

    def run(key: str) -> StageScore:
        return score_evidence(
            evidences[key], system_prompt_text=(prompts or {}).get(key)
        )

    with ThreadPoolExecutor(max_workers=max(1, workers)) as pool:
        # 제출은 한꺼번에, 회수는 단계 순서대로 한다(진행 표시가 목록 순서와 같아진다).
        futures = {key: pool.submit(run, key) for key in pending}
        for key, future in futures.items():
            try:
                results[key] = future.result()
                error: Exception | None = None
            except Exception as exc:  # noqa: BLE001 — 한 단계 실패가 전체를 막지 않는다
                error = exc
            if on_done is not None:
                on_done(key, results.get(key), error)
    return results


def composite(results: dict[str, StageScore] | list[StageScore]) -> dict[str, Any]:
    """종합 스코어(채점된 대상의 단순 평균)."""
    values = list(results.values()) if isinstance(results, dict) else list(results)
    return stage_schema.composite(values)

````


## app/analysis/stage_schema.py

````python
"""단계 채점 결과 계약과 총점 계산.

문서가 요구하는 것 둘을 코드가 보장한다.

1. **항목마다 점수와 이유.** 결과 모델을 배점표에서 만들기 때문에 항목이 하나라도 빠지면 스키마
   검증에서 실패한다. `reason` 은 빈 값을 허용하지 않는다(만점이어도 무엇을 확인했는지 적는다).
2. **항목별 점수의 합 = 총점.** 총점은 LLM 이 아니라 [tally][app.analysis.stage_schema.tally] 가
   계산한다. 평가할 입력이 없어 `applied=false` 인 항목의 배점은 문서 규칙대로 나머지 항목에
   비례 배분한다.

모델은 단계마다 다르므로 `create_model` 로 배점표에서 만들어 캐시한다. 항목 키가 곧 field 이름이라
provider 의 strict JSON Schema 가 항목 누락·오타를 막아 준다.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from functools import lru_cache
from typing import Any, Literal

from pydantic import BaseModel, Field, create_model, field_validator

from app.analysis import stages
from app.analysis.stages import StageSpec

Severity = Literal["HIGH", "MEDIUM", "LOW"]
AlertKind = Literal["HALLUCINATION", "MISSING_SOURCE", "SENSITIVE"]

ALERT_LABELS: dict[str, str] = {
    "HALLUCINATION": "입력에서 확인되지 않는 사실",
    "MISSING_SOURCE": "필수 반영 source 누락",
    "SENSITIVE": "민감정보 노출",
}
"""문서가 "점수와 별개로 명시"하라고 한 3종. 점수와 따로 모아 화면 맨 위에 띄운다."""


class ItemScore(BaseModel):
    """채점 항목 하나의 결과."""

    score: int = Field(description="이 항목에서 받은 점수(0 ~ 배점). 배점을 넘기지 않는다")
    applied: bool = Field(
        description="이 항목을 평가할 입력이 있었는지. 없으면 false 로 두고 감점하지 않는다"
    )
    reason: str = Field(

@dataclass(frozen=True)
class StageScore:
    """단계 하나의 최종 채점 결과. 화면·저장이 함께 쓰는 형태."""

    stage_key: str
    label: str
    total: float | None
    items: tuple[ScoredItem, ...]
    deductions: tuple[dict[str, Any], ...] = ()
    alerts: tuple[dict[str, Any], ...] = ()
    summary: str = ""
    skip_reason: str | None = None
    redistributed: tuple[str, ...] = ()
    """평가 입력이 없어 배점을 나눠준 항목 키."""

    extras: dict[str, Any] = field(default_factory=dict)
    """점수와 무관한 기록: `usage` — 대상 서버가 이 단계에 쓴 토큰·시간."""

    @property
    def evaluated(self) -> bool:
        return self.total is not None

    def to_dict(self) -> dict[str, Any]:
        return {
            "stageKey": self.stage_key,
            "label": self.label,
            "total": self.total,
            "items": [
                {
                    "key": item.key,
                    "label": item.label,
                    "score": item.score,
                    "points": item.points,
                    "appliedPoints": item.applied_points,
                    "applied": item.applied,
                    "reason": item.reason,
                }
                for item in self.items
            ],
            "deductions": list(self.deductions),
            "alerts": list(self.alerts),
            "summary": self.summary,
            "skipReason": self.skip_reason,
            "redistributed": list(self.redistributed),
            "usage": self.extras.get("usage") or {},
            "scoringPolicy": self.extras.get("scoringPolicy"),
            "checks": self.extras.get("checks") or {},
        }


def skipped(spec: StageSpec, reason: str, *, usage: dict[str, Any] | None = None) -> StageScore:
    """채점하지 않은 단계. 0점이 아니라 총점 없음으로 남긴다."""
    return StageScore(
        spec.key, spec.label, None, (), summary="", skip_reason=reason, extras={"usage": usage or {}}
    )


def tally(spec: StageSpec, card: BaseModel, *, extras: dict[str, Any] | None = None) -> StageScore:
    """LLM 결과 → 점수 계산이 끝난 결과.

    `applied=false` 항목의 배점은 나머지 항목에 비례 배분한다(문서 규칙). 그래서 총점은 항상
    100점 만점이고, 항목별 반영 배점의 합과 정확히 맞는다.
    """
    raw_items = card.items
    applied_points = sum(
        item.points for item in spec.items if getattr(raw_items, item.key).applied
    )
    scored: list[ScoredItem] = []
    redistributed: list[str] = []
    for item in spec.items:
        value = getattr(raw_items, item.key)
        score = max(0, min(item.points, int(value.score)))
        if not value.applied:
            redistributed.append(item.key)
            scored.append(ScoredItem(item.key, item.label, 0, item.points, 0.0, False, value.reason))
            continue
        share = round(item.points * stages.TOTAL_POINTS / applied_points, 2) if applied_points else 0.0
        scored.append(ScoredItem(item.key, item.label, score, item.points, share, True, value.reason))

    total = (
        round(
            sum(item.score for item in scored if item.applied) * stages.TOTAL_POINTS / applied_points,
            1,
        )
        if applied_points
        else None
    )
    return StageScore(
        stage_key=spec.key,
        label=spec.label,
        total=total,
        items=tuple(scored),
        deductions=tuple(item.model_dump() for item in card.deductions),
        alerts=tuple(item.model_dump() for item in card.alerts),
        summary=card.summary,
        skip_reason=None if applied_points else "모든 항목에 평가할 입력이 없었습니다.",
        redistributed=tuple(redistributed),
        extras=dict(extras or {}),
    )


def composite(scores: list[StageScore]) -> dict[str, Any]:
    """종합 스코어 = 채점된 평가 대상들의 단순 평균(문서 규칙). 미채점 대상은 제외한다."""
    evaluated = [score for score in scores if score.evaluated]
    return {
        "score": round(sum(score.total for score in evaluated) / len(evaluated), 1) if evaluated else None,
        "stageScores": {score.stage_key: score.total for score in evaluated},
        "skipped": {
            score.stage_key: score.skip_reason for score in scores if not score.evaluated
        },
        "evaluatedCount": len(evaluated),
    }
````

````python
"""단계별 채점 근거 슬라이서 — trace.json 에서 "그 단계가 본 것과 만든 것"만 잘라낸다.

원칙 넷:

1. **자르되 요약하지 않는다.** 남기기로 한 부분은 원문 그대로 넣는다. 요약하면 채점자가 근거를
   대조할 수 없다.
2. **그 단계의 입력만 넣는다.** 대상 서버는 Event Agent 5종에 수집 원본 전체를 똑같이 넘긴다.
   그대로 채점하면 "자기 source 만 보고 만들었는지"를 잴 수 없으므로 해당 source 배열만 남긴다.
3. **화면과 프롬프트가 같은 것을 본다.** [sections][app.analysis.evidence.StageEvidence.sections]
   하나로 프롬프트를 만들고 화면도 그린다. 채점 근거와 화면이 어긋날 수 없다.
4. **없으면 N/A.** 입력이 0건이거나 단계가 실행되지 않았으면 채점하지 않고 이유를 남긴다.
   0점이 아니다(문서 규칙: 평가할 입력이 없는 기준은 감점하지 않는다).

단계 식별은 관측치 이름이 아니라 `step.metadata`(agent·iteration·tool·phase)를 쓴다.
"""

from __future__ import annotations

import json
import re
from dataclasses import dataclass, field
from typing import Any

from app.analysis import stages
from app.analysis.stages import StageSpec
from app.tasks.trace_builder import resolve

_EVENT_AGENTS = {
    "event-location": ("location", ("stays", "movements")),
    "event-calendar": ("calendar", ("calendars",)),
    "event-notification": ("notification", ("notifications",)),
    "event-photo": ("photo", ("photos",)),
}
"""stageKey → (metadata.agent 값, 그 단계가 실제로 담당하는 source 배열)."""

_PROMPT_SECTION = re.compile(r"^\[(?P<name>[A-Za-z][A-Za-z0-9 _-]*)\][ \t]*$", re.MULTILINE)


@dataclass(frozen=True)
class EvidenceSection:
    """근거 한 덩어리. 프롬프트의 `[제목]` 블록이자 화면의 소제목 하나."""

    title: str
    note: str
    payload: Any

    @property
    def is_empty(self) -> bool:
        return self.payload in (None, [], {}, "")


@dataclass(frozen=True)
class StageEvidence:
    """단계 하나의 채점 근거. 채점할 수 없으면 `applicable=False` 와 이유만 들고 있다."""

    stage: StageSpec
    applicable: bool
    skip_reason: str | None = None
    summary: dict[str, Any] = field(default_factory=dict)
    sections: tuple[EvidenceSection, ...] = ()
    usage: dict[str, Any] = field(default_factory=dict)
    """대상 서버가 이 단계에 쓴 토큰·시간. 채점 점수와는 무관하고 기록용이다."""

    @property

        return self.stage.key


def _steps(trace_json: dict) -> list[dict]:
    return list((trace_json.get("process") or {}).get("steps") or [])

def _message_text(payload: Any, role: str) -> str | None:
    """messages 에서 특정 role 의 content 를 꺼낸다."""
    if not isinstance(payload, list):
        return None
    for message in payload:
        if isinstance(message, dict) and message.get("role") == role:
            return str(message.get("content", ""))
    return None


def prompt_section(text: str | None, name: str) -> str | None:
    """대상 서버 프롬프트의 `[블록]` 하나를 꺼낸다. 없으면 None.

    User Memory 원문은 수집 원본(`request.userMemory`)에 없고 프롬프트 안에만 있다. 블록 이름이
    바뀌면 못 찾는데, 그때는 해당 근거만 비고 채점은 계속된다(항목이 N/A 로 떨어진다).
    """
    if not text:
        return None
    matches = list(_PROMPT_SECTION.finditer(text))
    for index, match in enumerate(matches):
        if match.group("name").strip() != name:
            continue
        start = match.end()
        end = matches[index + 1].start() if index + 1 < len(matches) else len(text)
        body = text[start:end].strip()
        return body or None
    return None


def parse_json_array(value: Any) -> list[dict] | None:
    """events 블록 선두의 JSON 배열. 뒤의 자연어 실행 지시는 원문에 남기고 배열만 읽는다."""
    if isinstance(value, str):
        text = value.strip()
        try:
            value, end = json.JSONDecoder().raw_decode(text)
        except (ValueError, TypeError):
            return None
        if text[end:].lstrip().startswith(("[", "{", ",")):
            return None  # 여러 JSON 값을 임의로 하나로 축약하지 않는다.
    if not isinstance(value, list) or any(not isinstance(row, dict) for row in value):
        return None
    return value

def _event_agent_evidence(trace_json: dict, spec: StageSpec) -> StageEvidence:
    agent, source_keys = _EVENT_AGENTS[spec.key]
    step = _step_by_agent(trace_json, agent)
    source = trace_json.get("source") or {}
    counts = {key: len(source.get(key) or []) for key in source_keys}
    total = sum(counts.values())

    if step is None:
        return StageEvidence(spec, False, f"{spec.label} 관측치가 트레이스에 없습니다.", counts)
    if total == 0:
        return StageEvidence(spec, False, f"입력이 0건입니다({', '.join(source_keys)}).", counts)

    result = (step.get("output") or {}).get("result") or {}
    sections = [
        EvidenceSection("실행 정보", "이 단계가 언제·어떤 상태로 실행됐는지.", _header(trace_json, step)),
        EvidenceSection(
            "입력 source",
            f"이 단계가 담당하는 원본만 넣는다({', '.join(source_keys)}). 다른 source 는 이 단계의 근거가 아니다.",
            {key: source.get(key) or [] for key in source_keys},
        ),
    ]

    if spec.key == "event-photo":
        describe = _generation_by_name(trace_json, "describe-photo-images")
        sections.append(
            EvidenceSection(
                "Vision 결과",

                _messages(trace_json, describe.get("outputRef")) if describe else None,

        )

    sections.append(
        EvidenceSection(
            "출력 candidate·fragment·warning",
            "이 단계가 만든 것 전부.",
            {
                "candidates": result.get("candidates") or [],
                "fragments": result.get("fragments") or [],
                "warnings": result.get("warnings") or [],
            },
        )

    sections.extend(_prompt_sections(trace_json, next(iter(_generations_of(trace_json, step)), None), spec.label))

    summary = counts | {
        "candidates": len(result.get("candidates") or []),
        "fragments": len(result.get("fragments") or []),
        "warnings": len(result.get("warnings") or []),
    }

def _repair_iterations(trace_json: dict) -> list[dict]:
    """회차별 계획·도구 실행·결과. 회차 순서대로."""
    by_iteration: dict[Any, dict] = {}
    for step in _steps(trace_json):
        meta = step.get("metadata") or {}
        iteration = meta.get("iteration")
        if iteration is None:
            continue
        entry = by_iteration.setdefault(iteration, {"iteration": iteration, "toolCalls": []})
        name = step.get("name") or ""
        if name == "analyze-repair-iteration":
            entry["plan"] = (step.get("output") or {}).get("plan")
        elif name == "execute-repair-plan":
            entry["results"] = (step.get("output") or {}).get("results")
        elif name == "confirm-repair-iteration":
            entry["timelineRef"] = step.get("timelineRef")
    # execute-<tool> 은 iteration metadata 가 없고 execute-repair-plan 의 자식이다.
    plans = {
        step.get("id"): (step.get("metadata") or {}).get("iteration")
        for step in _steps(trace_json)
        if step.get("name") == "execute-repair-plan"
    }
    for step in _steps(trace_json):
        iteration = plans.get(step.get("parentId"))
        if iteration is None:
            continue
        by_iteration.setdefault(iteration, {"iteration": iteration, "toolCalls": []})["toolCalls"].append(
            {
                "tool": (step.get("metadata") or {}).get("tool"),
                "call": (step.get("input") or {}).get("call"),
                "level": step.get("level"),
                "result": (step.get("output") or {}).get("result"),
                "ok": (step.get("output") or {}).get("ok"),
            }
        )
    return [by_iteration[key] for key in sorted(by_iteration, key=lambda value: (value is None, value))]


def _repair_evidence(trace_json: dict, spec: StageSpec) -> StageEvidence:
    step = _step_by_agent(trace_json, "repair")
    if step is None:
        return StageEvidence(spec, False, "Repair Agent 관측치가 트레이스에 없습니다.")
    iterations = _repair_iterations(trace_json)
    if not any(entry.get("plan") for entry in iterations):
        return StageEvidence(spec, False, "Repair 반복이 한 번도 실행되지 않았습니다.")

    timeline_step = _step_by_agent(trace_json, "timeline")
    before = resolve(trace_json, (timeline_step or {}).get("timelineRef"))
    after = resolve(trace_json, step.get("timelineRef"))
    event_results = {
        (candidate.get("metadata") or {}).get("agent"): (candidate.get("output") or {}).get("result")
        for candidate in _steps(trace_json)
        if (candidate.get("metadata") or {}).get("sourceStage") == "event"
    }
    # 회차 종료본은 참조만 두면 화면에서 못 보므로, 회차별로 event 요약만 펼친다(전문 중복 방지).
    for entry in iterations:
        timeline = resolve(trace_json, entry.pop("timelineRef", None))
        entry["resultEvents"] = [
            {
                "clientEventId": event.get("clientEventId"),
                "eventType": event.get("eventType"),
                "title": event.get("title"),
                "startTime": event.get("startTime"),
                "endTime": event.get("endTime"),
            }
            for event in (timeline or {}).get("events") or []
        ]

    sections = (
        EvidenceSection("실행 정보", "이 단계가 언제·어떤 상태로 실행됐는지.", _header(trace_json, step)),
        EvidenceSection("수정 전 draft", "Repair 가 받은 타임라인. 문제 발견의 대상이다.", before),
        EvidenceSection(
            "근거 원본",
            "draft 주장을 검증할 수집 원본과 Agent 결과. lookup_source 없이 고칠 수 있었는지 여기서 본다.",
            {"source": trace_json.get("source") or {}, "eventResults": event_results},
        ),
        EvidenceSection(
            "회차별 계획과 도구 실행",
            "회차마다 issues·toolCalls·done·summary 와 실제 도구 호출·결과. 회차 종료본은 event 요약만 담았다.",
            iterations,
        ),
        EvidenceSection("수정 후 timeline", "Repair 가 끝내고 넘긴 타임라인 전부.", after),
        *_prompt_sections(
            trace_json, _generation_by_name(trace_json, "analyze-timeline-repair"), "Repair analyze"
        ),
    )
    summary = {
        "iterations": len(iterations),
        "issues": sum(len(((entry.get("plan") or {}).get("issues")) or []) for entry in iterations),
        "toolCalls": sum(len(entry.get("toolCalls") or []) for entry in iterations),
        "eventsBefore": len((before or {}).get("events") or []),
        "eventsAfter": len((after or {}).get("events") or []),
    }
    return StageEvidence(spec, True, None, summary, sections, stage_usage(step))


def _question_evidence(trace_json: dict, spec: StageSpec) -> StageEvidence:
    step = _step_by_agent(trace_json, "question")
    if step is None:
        return StageEvidence(spec, False, "Question Agent 관측치가 트레이스에 없습니다.")
    timeline = resolve(trace_json, step.get("timelineRef")) or {}
    events = timeline.get("events") or []
    generation = next(iter(_generations_of(trace_json, step)), None)
    user_prompt = _message_text(_messages(trace_json, (generation or {}).get("inputRef")), "user")
    input_events = prompt_section(user_prompt, "events")
    parsed_inputs = parse_json_array(input_events)
    if not events and not (isinstance(parsed_inputs, list) and parsed_inputs):
        return StageEvidence(spec, False, "질문을 붙일 event 가 0건이거나 입력 근거를 읽을 수 없습니다.")
    # 질문은 events[].question 에 embedded 된다. 예전 로그는 timeline.questions 배열이었다.
    embedded = [
        {"clientEventId": event.get("clientEventId"), "question": event.get("question")}
        for event in events
    ]
    questions = embedded if any(item["question"] for item in embedded) else (timeline.get("questions") or embedded)

    sections = (
        EvidenceSection("실행 정보", "이 단계가 언제·어떤 상태로 실행됐는지.", _header(trace_json, step)),
        EvidenceSection(
            "입력 event",
            "질문이 전제할 수 있는 사실의 전부. timeOfDay 는 프롬프트에만 있어 원문 그대로 넣는다.",

            "입력 User Memory",

            prompt_section(user_prompt, "user memory"),

    summary = {
        "events": len(parsed_inputs) if parsed_inputs is not None else len(events),
        "questions": sum(1 for item in questions or [] if (item or {}).get("question")),
    }
    return StageEvidence(spec, True, None, summary, sections, stage_usage(step))

def build(trace_json: dict, stage_key: str) -> StageEvidence:
    """단계 하나의 채점 근거. 채점할 수 없으면 `applicable=False` 로 돌아온다."""
    spec = stages.get(stage_key)
    trace_name = (trace_json.get("task") or {}).get("traceName")
    if trace_name and trace_name != spec.trace_name:
        return StageEvidence(
            spec, False, f"이 트레이스(`{trace_name}`)가 아니라 `{spec.trace_name}` 에서 채점하는 단계입니다."
        )
    builder = _BUILDERS.get(stage_key)
    if builder is not None:
        return builder(trace_json, spec)
    return _event_agent_evidence(trace_json, spec)


def build_all(trace_json: dict) -> list[StageEvidence]:
    """그 트레이스에서 다룰 수 있는 단계 전부(채점 불가 사유 포함)."""
    trace_name = (trace_json.get("task") or {}).get("traceName")
    specs = stages.for_trace(trace_name) or stages.STAGES
    return [build(trace_json, spec.key) for spec in specs]


def _as_text(payload: Any) -> str:
    if payload is None:
        return "(없음)"
    if isinstance(payload, str):
        return payload
    try:
        return json.dumps(payload, ensure_ascii=False, indent=2, default=str)
    except (TypeError, ValueError):
        return str(payload)


def build_user_prompt(evidence: StageEvidence) -> str:
    """근거를 채점 입력 메시지로 조립한다. 화면에 보이는 근거와 같은 내용·같은 순서다."""
    blocks = [
        f"평가 대상: {evidence.stage.label}",
        "아래 근거는 이 단계의 입력과 출력이다. 다른 단계의 결과는 채점 대상이 아니다.",
    ]
    for section in evidence.sections:
        blocks.append(f"[{section.title}] — {section.note}\n{_as_text(section.payload)}")
    blocks.append(
        "위 근거만으로 채점 기준의 **모든 항목**에 점수와 이유를 매겨라. 근거에 없는 사실을 지어내지 "
        "말고, 평가할 입력이 없는 항목은 `applied=false` 로 표시하라."
    )
    return "\n\n".join(blocks)
````

````python
"""채점 결과 레코드 — `data/evaluations/<taskId>.json` 의 계약.

Task 하나에 파일 하나다. 단계별로 따로 채점하므로 **한 단계를 채점하면 그 단계만 갱신**하고
나머지는 그대로 둔다. 종합 스코어는 저장할 때마다 남아 있는 단계 점수로 다시 계산한다.

```json
{
  "schemaVersion": 2,
  "taskId": "…", "traceId": "…", "env": "dev", "traceName": "generate-timeline",
  "composite": {"score": 84.6, "stageScores": {…}, "skipped": {…}},
  "stages": {
    "timeline": {
      "rubricRef": {"name": "judge-stage-timeline", "version": 1},
      "judgedAt": "…", "evidence": {"summary": {…}, "sections": ["실행 정보", …]},
      "scorecard": {"total": 78.0, "items": [...], "deductions": [...], "alerts": [...]}
    }
  }
}
```

근거 원문은 여기 담지 않는다. 정본은 `data/tasks/<taskId>/trace.json` 이고, 화면은 그것으로 근거를
다시 만들어 보여준다(같은 근거를 두 곳에 두면 어긋난다).

`schemaVersion` 이 없는 예전 레코드(7기준 0~10점 단일 카드)는 **읽기 전용**으로만 다룬다.

이 파일은 **최신 상태**만 들고 있다. 같은 단계를 여러 번 채점하면 실행마다 한 줄씩
`data/evaluations/history/<taskId>.jsonl` 에 쌓인다(→ [append_history][app.analysis.evaluation_record.append_history]).
채점자는 실행마다 점수가 달라지므로 편차를 보려면 이력이 필요하다.
"""

from __future__ import annotations

import json
from datetime import datetime, timezone
from pathlib import Path
from typing import Any
from uuid import uuid4

from app.analysis import rubric_store, stage_schema
from app.analysis.stage_schema import StageScore
from app.storage import store
from app.storage.paths import (
    evaluation_file,
    evaluation_history_dir,
    evaluation_history_file,
    evaluations_dir,
)

SCHEMA_VERSION = 2


def now_iso() -> str:
    return datetime.now(timezone.utc).astimezone().isoformat(timespec="seconds")


def is_legacy(record: dict[str, Any]) -> bool:
    """구버전(단일 scorecard) 레코드인지. 재채점·비교 대상에서 뺀다."""
    return int(record.get("schemaVersion") or 1) < SCHEMA_VERSION


def load(task_id: str) -> dict[str, Any] | None:
    data = store.load_json(evaluation_file(task_id))
    return data if isinstance(data, dict) else None


def load_all() -> list[tuple[Path, dict[str, Any]]]:
    """저장된 채점 결과 전부(최근 수정 순). 구버전 레코드도 그대로 읽는다."""
    records: list[tuple[Path, dict[str, Any]]] = []
    for path in store.list_json(evaluations_dir()):
        data = store.load_json(path)
        if isinstance(data, dict):
            records.append((path, data))
    return sorted(records, key=lambda item: item[0].stat().st_mtime, reverse=True)


def _recompute_composite(record: dict[str, Any]) -> dict[str, Any]:
    """저장된 단계 점수로 종합 스코어를 다시 계산한다(채점된 대상의 단순 평균)."""
    totals = {
        key: (entry.get("scorecard") or {}).get("total")
        for key, entry in (record.get("stages") or {}).items()
    }
    evaluated = {key: value for key, value in totals.items() if isinstance(value, (int, float))}
    skipped = {
        key: ((entry.get("scorecard") or {}).get("skipReason") or "채점하지 않음")
        for key, entry in (record.get("stages") or {}).items()
        if not isinstance(totals.get(key), (int, float))
    }
    return {
        "score": round(sum(evaluated.values()) / len(evaluated), 1) if evaluated else None,
        "stageScores": evaluated,
        "skipped": skipped,
        "evaluatedCount": len(evaluated),
    }


def save_stages(
    *,
    task_id: str,
    trace_id: str | None,
    env: str,
    trace_name: str | None,
    date: str | None,
    results: dict[str, StageScore],
    rubric_refs: dict[str, dict[str, Any]],
    evidence_summaries: dict[str, dict[str, Any]],
    trace_meta: dict[str, Any] | None = None,
) -> dict[str, Any]:
    """단계 채점 결과를 레코드에 합쳐 저장한다. 준 단계만 갱신한다."""
    current = load(task_id)
    record: dict[str, Any]
    if current is None or is_legacy(current):
        # 구버전 레코드는 덮어쓰지 않고 새 구조로 시작한다(구버전 파일은 legacy 키에 남긴다).
        record = {"schemaVersion": SCHEMA_VERSION, "stages": {}}
        if current is not None:
            record["legacy"] = current
    else:
        record = dict(current)
        record.setdefault("stages", {})

    record.update(
        {
            "taskId": task_id,
            "traceId": trace_id,
            "env": env,
            "traceName": trace_name,
            "name": trace_name,  # 기존 화면이 쓰던 키를 유지한다
            "date": date,
            "updatedAt": now_iso(),
        }
    )
    for key, score in results.items():
        record["stages"][key] = {
            "rubricRef": rubric_refs.get(key),
            "judgedAt": now_iso(),
            "evidence": evidence_summaries.get(key) or {},
            "scorecard": score.to_dict(),
        }
    record["composite"] = _recompute_composite(record)
    store.save_json(evaluation_file(task_id), record)
    if results:
        append_history(
            task_id=task_id,
            trace_id=trace_id,
            env=env,
            trace_name=trace_name,
            date=date,
            results=results,
            rubric_refs=rubric_refs,
            trace_meta=trace_meta,
        )
    return record


def append_history(
    *,
    task_id: str,
    trace_id: str | None,
    env: str,
    trace_name: str | None,
    date: str | None,
    results: dict[str, StageScore],
    rubric_refs: dict[str, dict[str, Any]],
    trace_meta: dict[str, Any] | None = None,
) -> str:
    """채점 실행 하나를 이력에 덧붙인다(한 줄 = 단계 1회 채점).

    같은 단계를 여러 번 채점하면 최신 결과만 `evaluation_file` 에 남지만, 이력에는 실행마다 한 줄씩
    쌓인다. 채점자가 실행마다 다른 점수를 내므로 편차를 보려면 이 이력이 필요하다.
    """
    run_id = uuid4().hex[:12]
    stamp = now_iso()
    path = evaluation_history_file(task_id)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as file:
        for key, score in results.items():
            file.write(
                json.dumps(
                    {
                        "runId": run_id,
                        "judgedAt": stamp,
                        "taskId": task_id,
                        "traceId": trace_id,
                        "env": env,
                        "traceName": trace_name,
                        "date": date,
                        "stageKey": key,
                        "label": score.label,
                        "total": score.total,
                        "rubricRef": rubric_refs.get(key),
                        "rubricEdited": rubric_store.is_edited(key),
                        # 점수와 무관한 기록: 대상 서버가 이 단계에 쓴 토큰·시간과 Task 전체 비용
                        "usage": score.extras.get("usage") or {},
                        "traceCost": (trace_meta or {}).get("cost"),
                        "traceLatency": (trace_meta or {}).get("latency"),
                        "traceTokens": (trace_meta or {}).get("tokens"),
                        "scorecard": score.to_dict(),
                    },
                    ensure_ascii=False,
                )
                + "\n"
            )
    return run_id


def load_history(task_id: str | None = None) -> list[dict[str, Any]]:
    """채점 실행 이력(최신순). `task_id` 를 주면 그 Task 만."""
    files = (
        [evaluation_history_file(task_id)]
        if task_id
        else sorted(evaluation_history_dir().glob("*.jsonl"))
    )
    runs: list[dict[str, Any]] = []
    for path in files:
        if not path.exists():
            continue
        for line in path.read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if not line:
                continue
            try:
                runs.append(json.loads(line))
            except json.JSONDecodeError:
                continue  # 저장 중 끊긴 줄은 건너뛴다
    return sorted(runs, key=lambda run: run.get("judgedAt") or "", reverse=True)


def findings_text(record: dict[str, Any]) -> str:
    """단계별 감점 근거를 사람이 읽는 한 덩어리로 만든다.

    개선책·프롬프트 제안·테스트 데이터 생성이 같은 형식을 쓴다. **평가 대상(Agent)별로** 묶여
    있어 "어느 단계의 프롬프트를 고쳐야 하는지"가 그대로 드러난다.
    """
    lines = [f"## task {record.get('taskId')} (종합 {composite_score(record)}/100)"]
    for key, entry in (record.get("stages") or {}).items():
        card = entry.get("scorecard") or {}
        if card.get("total") is None:
            continue
        label = card.get("label") or key
        lines.append(f"### {label} {card['total']}/100 — {card.get('summary') or ''}")
        for alert in card.get("alerts") or []:
            lines.append(f"- [ALERT/{alert.get('kind')}] {alert.get('where')} — {alert.get('detail')}")
        for item in card.get("deductions") or []:
            refs = ", ".join(item.get("sourceRefs") or []) or "-"
            lines.append(
                f"- [{item.get('severity')}] ({item.get('itemKey')}) {item.get('description')} "
                f"· 입력 {refs} · 출력 {item.get('outputPath')}"
            )
    return "\n".join(lines)


def stage_score(record: dict[str, Any], stage_key: str) -> dict[str, Any] | None:
    """저장된 단계 채점 결과 하나."""
    entry = (record.get("stages") or {}).get(stage_key)
    return entry if isinstance(entry, dict) else None


def composite_score(record: dict[str, Any]) -> float | None:
    """종합 스코어. 구버전 레코드는 0~10 점수를 100점으로 환산해 비교 가능하게 만든다."""
    if is_legacy(record):
        legacy = ((record.get("scorecard") or {}).get("overall") or {}).get("score")
        return round(float(legacy) * 10, 1) if isinstance(legacy, (int, float)) else None
    return (record.get("composite") or {}).get("score")


def to_stage_scores(record: dict[str, Any]) -> list[StageScore]:
    """저장된 레코드 → 계산용 StageScore 목록(집계·비교에 쓴다)."""
    scores: list[StageScore] = []
    for key, entry in (record.get("stages") or {}).items():
        card = entry.get("scorecard") or {}
        scores.append(
            StageScore(
                stage_key=key,
                label=card.get("label") or key,
                total=card.get("total"),
                items=tuple(
                    stage_schema.ScoredItem(
                        key=item.get("key"),
                        label=item.get("label"),
                        score=item.get("score") or 0,
                        points=item.get("points") or 0,
                        applied_points=item.get("appliedPoints") or 0.0,
                        applied=bool(item.get("applied")),
                        reason=item.get("reason") or "",
                    )
                    for item in card.get("items") or []
                ),
                deductions=tuple(card.get("deductions") or []),
                alerts=tuple(card.get("alerts") or []),
                summary=card.get("summary") or "",
                skip_reason=card.get("skipReason"),
                extras={
                    "usage": card.get("usage") or {},
                    "scoringPolicy": card.get("scoringPolicy"),
                    "checks": card.get("checks") or {},
                },
            )
        )
    return scores
````

````python
def run():
    parser = argparse.ArgumentParser()
    parser.add_argument("--model", required=True)
    parser.add_argument("--repeats", type=int, default=5)
    parser.add_argument("--workers", type=int, default=2)
    args = parser.parse_args()
    if args.repeats < 1:
        parser.error("repeats는 1 이상이어야 합니다")
    baseline = registry.get("judge-stage-question", 2)
    if not baseline:
        raise SystemExit("기존 judge-stage-question v2 원문이 없어 비교를 중단합니다")
    new_prompt = stage_judge.system_prompt("question")
    provider = CodexProvider(model=args.model)
    output_dir = Path("data/validation") / datetime.now().strftime("%Y%m%d-%H%M%S")
    output_dir.mkdir(parents=True, exist_ok=False)
    cases = dict(fixtures())
    store.save_json(output_dir / "setup.json", {
        "model": args.model, "repeats": args.repeats, "stage": "question",
        "baselineVersion": 2, "baselinePrompt": baseline["content"], "newPrompt": new_prompt,
        "inputs": {name: build_user_prompt(ev) for name, ev in cases.items()},
        "promptHashes": {name: hashlib.sha256(text.encode()).hexdigest() for name, text in (("old", baseline["content"]), ("new", new_prompt))},
        "expectedCoverage": {"heldout-good": 20, "heldout-missing-and-unsupported": 15},
        "expectedUnsupportedRef": {"heldout-good": None, "heldout-missing-and-unsupported": "E2"},
        "limitation": "합성 보류 사례 2개. 전체 점수의 사람 확정 정답은 없으며 실제 사용자 데이터 분포를 대표하지 않음.",
    })

    def judge(name, variant, iteration):
        ev = cases[name]
        try:
            if variant == "new":
                result = stage_judge.score_evidence(ev, system_prompt_text=new_prompt)
            else:
                card = provider.evaluate(system_prompt=baseline["content"], user_prompt=build_user_prompt(ev), result_model=stage_schema.scorecard_model("question"))
                result = stage_schema.tally(ev.stage, card)
            entry = {"case": name, "variant": variant, "iteration": iteration, "scorecard": result.to_dict()}
        except Exception as exc:
            entry = {"case": name, "variant": variant, "iteration": iteration, "error": f"{type(exc).__name__}: {exc}"}
        store.save_json(output_dir / f"{name}-{variant}-{iteration}.json", entry)
        return entry

    results = []
    with patch.object(stage_judge, "get_llm_provider", return_value=provider), ThreadPoolExecutor(max_workers=args.workers) as pool:
        jobs = [pool.submit(judge, name, variant, iteration) for iteration in range(1, args.repeats+1) for name in cases for variant in ("old", "new")]
        for job in as_completed(jobs):
            entry = job.result()
            results.append(entry)
            print(json.dumps({key: entry[key] for key in ("case", "variant", "iteration")} | {"total": entry.get("scorecard", {}).get("total"), "error": entry.get("error")}, ensure_ascii=False), flush=True)
    summary = []
    for name in cases:
        for variant in ("old", "new"):
            entries = [row for row in results if row["case"] == name and row["variant"] == variant]
            cards = [row["scorecard"] for row in entries if "scorecard" in row]
            totals = [row["total"] for row in cards if row["total"] is not None]
            item_scores = {item.key: [next(x["score"] for x in card["items"] if x["key"] == item.key) for card in cards] for item in stages.get("question").items}
            summary.append({"case": name, "variant": variant, "successes": len(cards), "failures": len(entries)-len(cards),
                            "totals": totals, "range": max(totals)-min(totals) if totals else None,
                            "stddev": statistics.pstdev(totals) if totals else None, "itemScores": item_scores,
                            "naCounts": [len(card["redistributed"]) for card in cards]})
    store.save_json(output_dir / "summary.json", {"model": args.model, "summary": summary})
    print(f"RESULT_DIR={output_dir.resolve()}", flush=True)


if __name__ == "__main__":
    run()
````

````python
def assessment(total, passed):
    return calibration.CheckAssessment(
        total=total, passed=passed, failedRefs=[f"E{i}" for i in range(total-passed)], reason="검증 근거",
    )


class CalibrationTests(unittest.TestCase):
    def test_fixed_formula_and_na(self):
        cases = [([assessment(5, 5), assessment(5, 5)], (20, True)),
                 ([assessment(5, 4), assessment(5, 5)], (18, True)),
                 ([assessment(5, 4), assessment(0, 0)], (16, True)),
                 ([assessment(5, 0)], (0, True)), ([assessment(0, 0)], (0, False))]
        for rows, expected in cases:
            self.assertEqual(calibration.points_from_checks(20, rows), expected)
        self.assertEqual(calibration.points_from_checks(5, [assessment(2, 1)]), (3, True))

    def test_invalid_counts_and_refs_rejected(self):
        for changes in ({"passed": 2}, {"total": -1}, {"failedRefs": []}, {"reason": " "}, {"total": True}):
            with self.assertRaises(ValidationError):
                calibration.CheckAssessment.model_validate({"total": 1, "passed": 0, "failedRefs": ["E1"], "reason": "r", **changes})
````
